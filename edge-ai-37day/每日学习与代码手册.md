```python
def fixtures(directory):
    directory=Path(directory);directory.mkdir(parents=True,exist_ok=True)
    records=[]
    for frame,(w,h) in enumerate([(320,180),(641,359),(96,160),(321,241),(640,360),(65,65)]):
        x0,y0,x1,y1=w//5,h//4,w//2,h//2
        pixels=bytearray(w*h*3)
        for y in range(y0,y1):
            for x in range(x0,x1): pixels[(y*w+x)*3]=255
        path=directory/f'{frame:05}.ppm'
        path.write_bytes(f'P6\n{w} {h}\n255\n'.encode()+pixels)
        records.append(dict(file=path.name,width=w,height=h,expected_box=[x0,y0,x1,y1]))
    (directory/'manifest.json').write_text(json.dumps(records,indent=2),encoding='utf-8')
    return records
```

**按这个思路读：** bytearray 创建像素内存，三层索引决定哪一个颜色分量变红；manifest 保存期望框，后面用它检查算法。

### 今天重点看懂的实现

```python
def export_onnx(args):
    import torch
    import onnx
    import onnxruntime as ort
    import numpy as np
    from PIL import Image
    model = RTDetrBackend(args.weights)
    with Image.open(image_path(args.image)) as image:
        sample = model.tensor(image.convert('RGB'))
    module = model.export_module()
    MODELS.mkdir(parents=True, exist_ok=True)
    dest = MODELS/'detector.onnx'
    with torch.inference_mode():
        expected = module(sample).cpu().numpy()
        torch.onnx.export(module, sample, str(dest), input_names=[INPUT_NAME],
                          output_names=[OUTPUT_NAME], opset_version=17,
                          dynamic_axes=None, do_constant_folding=True, dynamo=False)
    onnx.checker.check_model(onnx.load(str(dest)))
    session = ort.InferenceSession(str(dest), providers=['CPUExecutionProvider'])
    actual = session.run([OUTPUT_NAME], {INPUT_NAME: sample.cpu().numpy()})[0]
    if actual.shape != (1, 300, 6):
        raise RuntimeError(f'Unexpected deployment output {actual.shape}')
    # Compare detections rather than row order: top-k ordering can change near ties.
    report = compare_boxes(decode_rows(expected[0], 640, 640), decode_rows(actual[0], 640, 640))
    save_json(MODELS/'onnx-parity.json', report)
    save_json(MODELS/'onnx_manifest.json', dict(
        contract=CONTRACT, model=args.weights, input=[1, 3, 640, 640], output=[1, 300, 6],
        names=model.names, sha256=hashlib.sha256(dest.read_bytes()).hexdigest(),
        onnx=onnx.__version__, onnxruntime=ort.__version__))
    print(dest)
```

**按这个思路读：** 导出 wrapper 定义契约；checker 检查图结构，ORT 实际执行后逐框比较；Day06/17 再验证 TRT 与 C++。

```cpp
class Pool {
    enum State {FREE,FILLING,READY,READING};
    struct Slot {Image image;uint64_t generation=0;State state=FREE;};
    std::vector<Slot> slots; mutable std::mutex mutex;
    Slot& check(Lease l){auto&s=slots.at(l.slot);if(s.generation!=l.generation||s.state==FREE)throw std::runtime_error("stale/early-recycled lease");return s;}
public:
    explicit Pool(int n):slots(n){if(n<1)throw std::runtime_error("pool must be positive");}
    std::optional<Lease> acquire(){std::lock_guard<std::mutex> lock(mutex);for(size_t i=0;i<slots.size();i++)if(slots[i].state==FREE){auto&s=slots[i];s.state=FILLING;return Lease{int(i),++s.generation};}return {};}
    Image& writable(Lease l){std::lock_guard<std::mutex>lock(mutex);auto&s=check(l);if(s.state!=FILLING)throw std::runtime_error("not filling");return s.image;}
    void publish(Lease l){std::lock_guard<std::mutex>lock(mutex);auto&s=check(l);if(s.state!=FILLING)throw std::runtime_error("not filling");s.state=READY;}
    void fill(Lease l,Image image){std::lock_guard<std::mutex> lock(mutex);auto&s=check(l);if(s.state!=FILLING)throw std::runtime_error("not filling");s.image=std::move(image);s.state=READY;}
    View read(Lease l){std::lock_guard<std::mutex> lock(mutex);auto&s=check(l);if(s.state!=READY)throw std::runtime_error("not ready");s.state=READING;return {s.image.rgb.data(),s.image.w,s.image.h};}
    void release(Lease l){std::lock_guard<std::mutex> lock(mutex);check(l).state=FREE;}
    bool all_free()const{std::lock_guard<std::mutex>lock(mutex);for(auto&s:slots)if(s.state!=FREE)return false;return true;}
};
```
