cmake_minimum_required(VERSION 3.18)
project(edge_ai_camera LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(EDGE_WITH_JETSON "Enable NVIDIA surface/EGL shared camera backend" OFF)
add_subdirectory(../pc pc)
add_executable(capture_probe capture_probe.cpp)
target_include_directories(capture_probe PRIVATE ../pc/include)
add_executable(edge_camera camera_run.cpp)
target_link_libraries(edge_camera PRIVATE edge_core)
if(EDGE_WITH_JETSON)
  find_package(CUDAToolkit REQUIRED)
  if(NOT EDGE_WITH_CUDA)
    message(FATAL_ERROR "Shared camera path requires EDGE_WITH_CUDA=ON")
  endif()
  find_path(NVBUF_INCLUDE nvbufsurface.h HINTS /usr/src/jetson_multimedia_api/include REQUIRED)
  find_library(NVBUF_LIB nvbufsurface HINTS /usr/lib/aarch64-linux-gnu/nvidia /usr/lib/aarch64-linux-gnu/tegra REQUIRED)
  find_library(EGL_LIB EGL REQUIRED)
  target_include_directories(edge_camera PRIVATE ${NVBUF_INCLUDE})
  target_compile_definitions(edge_camera PRIVATE EDGE_JETSON)
  target_link_libraries(edge_camera PRIVATE ${NVBUF_LIB} ${EGL_LIB} CUDA::cuda_driver CUDA::cudart)
endif()
