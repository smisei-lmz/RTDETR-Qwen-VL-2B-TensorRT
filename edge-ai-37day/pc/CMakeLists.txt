cmake_minimum_required(VERSION 3.18)
project(edge_ai_pc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(EDGE_WITH_CUDA "Build CUDA preprocessing and asynchronous scheduler" OFF)
option(EDGE_WITH_TENSORRT "Build TensorRT RT-DETRv2 top-k backend" OFF)
find_package(Threads REQUIRED)
add_library(edge_core src/cpu.cpp)
target_include_directories(edge_core PUBLIC include)
target_link_libraries(edge_core PUBLIC Threads::Threads)
if(EDGE_WITH_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(edge_core PRIVATE src/cuda_backend.cu)
  target_compile_definitions(edge_core PUBLIC EDGE_CUDA)
  target_link_libraries(edge_core PUBLIC CUDA::cudart)
  set_target_properties(edge_core PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
if(EDGE_WITH_TENSORRT)
  if(NOT EDGE_WITH_CUDA)
    message(FATAL_ERROR "TensorRT requires EDGE_WITH_CUDA")
  endif()
  find_path(TRT_INCLUDE NvInfer.h HINTS /usr/include/x86_64-linux-gnu /usr/include/aarch64-linux-gnu $ENV{TENSORRT_ROOT}/include REQUIRED)
  find_library(TRT_LIB nvinfer HINTS /usr/lib/x86_64-linux-gnu /usr/lib/aarch64-linux-gnu $ENV{TENSORRT_ROOT}/lib REQUIRED)
  target_include_directories(edge_core PRIVATE ${TRT_INCLUDE})
  target_link_libraries(edge_core PUBLIC ${TRT_LIB})
  target_compile_definitions(edge_core PUBLIC EDGE_TENSORRT)
endif()
add_executable(edge_pipeline src/main.cpp)
target_link_libraries(edge_pipeline PRIVATE edge_core)
add_executable(core_tests src/core_tests.cpp)
target_link_libraries(core_tests PRIVATE edge_core)
enable_testing()
add_test(NAME core_tests COMMAND core_tests)
