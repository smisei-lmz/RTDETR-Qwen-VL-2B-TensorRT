cmake_minimum_required(VERSION 3.22)
# Retain the CUDA toolkit used by the existing YOLO project; allow overrides.
if(NOT DEFINED CMAKE_CUDA_COMPILER AND EXISTS "/usr/local/cuda-13.4/bin/nvcc")
    set(CMAKE_CUDA_COMPILER "/usr/local/cuda-13.4/bin/nvcc" CACHE FILEPATH "CUDA compiler")
endif()
project(rtdetr_tensorrt LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(TENSORRT_ROOT "" CACHE PATH "Optional TensorRT 10.13.3 SDK root (include/ and lib/)")
set(TENSORRT_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/TensorRT-10.13.3/include" CACHE PATH "TensorRT headers")
set(TENSORRT_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../.venv-trt10/lib/python3.10/site-packages/tensorrt_libs/libnvinfer.so.10" CACHE FILEPATH "TensorRT 10 runtime")
if(TENSORRT_ROOT)
    set(TENSORRT_INCLUDE_DIR "${TENSORRT_ROOT}/include")
    find_library(TRT_SDK_LIBRARY NAMES nvinfer libnvinfer.so.10 PATHS "${TENSORRT_ROOT}/lib" "${TENSORRT_ROOT}/lib64" NO_DEFAULT_PATH REQUIRED)
    set(TENSORRT_LIBRARY "${TRT_SDK_LIBRARY}")
endif()
if(NOT EXISTS "${TENSORRT_INCLUDE_DIR}/NvInfer.h" OR NOT EXISTS "${TENSORRT_LIBRARY}")
    message(FATAL_ERROR "Missing TensorRT 10 headers/runtime. Set TENSORRT_ROOT or TENSORRT_INCLUDE_DIR and TENSORRT_LIBRARY.")
endif()
file(READ "${TENSORRT_INCLUDE_DIR}/NvInferVersion.h" TRT_VERSION_HEADER)
if(NOT TRT_VERSION_HEADER MATCHES "#define (NV_TENSORRT_MAJOR|TRT_MAJOR_ENTERPRISE)[ \t]+10([^0-9]|$)")
    message(FATAL_ERROR "Use TensorRT 10 headers with the existing TensorRT 10 engines")
endif()
get_filename_component(TENSORRT_LIBRARY "${TENSORRT_LIBRARY}" ABSOLUTE)
get_filename_component(TENSORRT_LIB_DIR "${TENSORRT_LIBRARY}" DIRECTORY)
find_package(CUDAToolkit REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
message(STATUS "TensorRT headers: ${TENSORRT_INCLUDE_DIR}")
message(STATUS "TensorRT runtime: ${TENSORRT_LIBRARY}")

set(TRT_SOURCES src/BufferManager.cpp src/Preprocess.cpp src/Postprocess.cpp
    src/TrtEngine.cpp src/load_TrtEngine_runtime.cpp src/laneManager.cpp
    src/CudaPreprocess.cu src/CudaPostprocess.cu)
add_executable(trt_example main.cpp ${TRT_SOURCES})
add_executable(trt_benchmark benchmark.cpp ${TRT_SOURCES})
add_executable(trt_lane_check lane_check.cpp ${TRT_SOURCES})
foreach(target trt_example trt_benchmark trt_lane_check)
    target_include_directories(${target} PRIVATE include ${OpenCV_INCLUDE_DIRS})
    target_include_directories(${target} SYSTEM PRIVATE "${TENSORRT_INCLUDE_DIR}")
    target_link_libraries(${target} PRIVATE "${TENSORRT_LIBRARY}" CUDA::cudart ${OpenCV_LIBS} ${CMAKE_DL_LIBS})
    target_compile_definitions(${target} PRIVATE RTDETR_PROJECT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/..")
    set_target_properties(${target} PROPERTIES BUILD_RPATH "${TENSORRT_LIB_DIR}" CUDA_ARCHITECTURES OFF)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-arch=native>)
endforeach()
